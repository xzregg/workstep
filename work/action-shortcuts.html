#!/usr/bin/env bash
set -euo pipefail

worktrees_file="${WORKSTEP_WORKTREES_FILE}"
backend_dir="$(jq -r '.worktrees[] | select(.repository_id == "repo-backend") | .path' "$worktrees_file")"
web_dir="$(jq -r '.worktrees[] | select(.repository_id == "repo-web") | .path' "$worktrees_file")"

cleanup() {
  kill "$backend_pid" "$web_pid" 2>/dev/null || true
}
trap cleanup TERM INT EXIT

(cd "$backend_dir" && ./scripts/start.sh) &
backend_pid=$!
(cd "$web_dir" && yarn dev) &
web_pid=$!
wait
